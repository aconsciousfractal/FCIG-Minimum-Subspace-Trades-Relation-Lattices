\section{A global localized theorem for the \texorpdfstring{\(\rh/\cB_1\)}{rho/B1} component}
\label{sec:global-rho}

Put
\[
 R=\ZZ[\zeta],\qquad \zeta=\zeta_{17},\qquad
 R_S=R\left[\frac1{2\cdot3\cdot17}\right].
\]
The power basis \(1,\zeta,\ldots,\zeta^{15}\) is an integral basis of
\(R\) \cite{Washington1997}.  Let
\[
 E=E_{\rh,1}^{\mathrm{full}}
\]
be the matrix whose rows are all complete minimum-circuit equations and
whose \(1317\) columns are the fixed \(\rh/\cB_1\) multiplicity
coordinates.  The five-dimensional point-induced subspace lies in the
kernel of every circuit equation.  Hence
\[
 \rank E\leq1317-5=1312
\]
after every scalar extension.  We write \(I_{1312}(E)\subseteq R\) for
the ideal generated by all order-\(1312\) minors of this complete
equation matrix.

\subsection{Exact lift and determinant reconstruction}

\begin{lemma}[Common cyclotomic lift]
\label{lem:common-cyclotomic-lift}
The \(1317\) source columns and every complete minimum-circuit equation
for \(E\) have one exact realization over \(R\) in the power basis above.
Under a residue homomorphism sending \(\zeta\) to an element of order
\(17\), this realization reduces entrywise to the corresponding complete
finite component equations.
\end{lemma}

\begin{proof}
The fixed-space columns are normalized as exponent vectors modulo \(17\).
Every representation entry is a signed sum of those powers, reduced by
\[
 1+\zeta+\cdots+\zeta^{16}=0.
\]
Circuit equations are formed from the same six signed block columns
before specialization.  Thus evaluation at a seventeenth root commutes
with column construction and row formation.
\end{proof}

\begin{lemma}[Certified signed CRT reconstruction]
\label{lem:certified-crt}
Let \(B\) be either square matrix in
\cref{prop:two-exact-minors}, and write
\[
 \det B=\sum_{j=0}^{15}c_j\zeta^j.
\]
Hadamard's inequality and Fourier inversion give the certified bound
\[
 \max_j|c_j|
 \leq\frac{32}{17}
 \max_{\sigma:R\hookrightarrow\CC}|\sigma(\det B)|.
\]
If the product \(M\) of the recorded split CRT primes is larger than
twice this bound, the signed coefficients \(c_j\) are uniquely determined
by their residues modulo \(M\).
\end{lemma}

\begin{proof}
For each embedding, Hadamard bounds the determinant by the square root of
the product of the certified row energies.  Applying the inverse
seventeenth-root transform and eliminating the coefficient of
\(\zeta^{16}\) gives the displayed bound.  Uniqueness follows
coordinatewise in the signed interval \((-M/2,M/2)\).
\end{proof}

\begin{proposition}[Two exact order-\(1312\) minors]
\label{prop:two-exact-minors}
The complete equation matrix \(E\) contains two explicitly specified
\(1312\times1312\) submatrices, using different equation-row bases and
different \(1312\)-column selections, with determinants
\(\delta_{\mathrm G},\delta_{\mathrm R}\in R\).  Their frozen exact
reconstruction data are
\[
\begin{array}{lrrrrrr}
\toprule
\text{minor}&\text{entries}&\text{direct check}&\text{CRT primes}&
\text{bound bits}&\text{modulus bits}&\text{norm bits}\\
\midrule
\delta_{\mathrm G}&7830&89\bmod103&96&2937&2967&4833\\
\delta_{\mathrm R}&7819&107\bmod137&93&2872&2874&2003\\
\bottomrule
\end{array}
\]
The largest reconstructed coefficient has respectively \(304\) and
\(128\) bits.
\end{proposition}

\begin{proof}
For \(\delta_{\mathrm G}\), a complete equation-row basis at
\((103,\zeta-8)\), together with its specified column selection, gives
determinant \(89\).  Its exact sparse lift has \(7830\) nonzero entries.
For \(\delta_{\mathrm R}\), reverse canonical traversal of the complete
circuit equations supplies a different row basis; with columns
\(0,\ldots,1311\), its determinant is \(107\) at
\((137,\zeta-122)\), and its exact lift has \(7819\) nonzero entries.

Standalone binary parsers directly reconstruct and check these two
diagnostic determinants.  The public reconstruction verifier separately
audits the frozen CRT prime lists, every recorded coefficient
congruence, the row-energy bounds, signed uniqueness, the exact
cyclotomic norms, and mutation controls.  The CRT determinant-generation
transcript itself is a source-admitted certificate rather than a
clean-room recomputation of all \(189\) modular determinants.
\end{proof}

\subsection{The determinantal-ideal certificate}

\begin{lemma}[Base change for determinantal ideals]
\label{lem:determinantal-base-change}
For a matrix \(M\) over a commutative ring \(T\), a ring map
\(T\to T'\), and an integer \(r\),
\[
 I_r(M)T'=I_r(M\otimes_TT').
\]
If \(I_r(M)T'=T'\), then \(M\) has rank at least \(r\) over every local
ring of \(T'\).
\end{lemma}

\begin{proof}
Both statements follow by applying the ring map to the determinants of
all \(r\times r\) submatrices.  If the extended ideal is the unit ideal,
at least one such determinant is a unit after localization at each prime.
\end{proof}

\begin{lemma}[Multiplication-lattice certificate]
\label{lem:multiplication-lattice}
For \(\alpha\in R\), let \(M(\alpha)\in\operatorname{Mat}_{16}(\ZZ)\)
be multiplication by \(\alpha\) in the integral power basis.  Then the
columns of
\[
 [\,M(\alpha)\ M(\beta)\,]\in\operatorname{Mat}_{16\times32}(\ZZ)
\]
span \(\alpha R+\beta R\), and its nonzero Smith entries are the additive
invariant factors of \(R/(\alpha R+\beta R)\).
\end{lemma}

\begin{proof}
The columns of \(M(\alpha)\) are
\(\alpha,\alpha\zeta,\ldots,\alpha\zeta^{15}\) in coordinates.
Concatenation adds the two principal-ideal lattices, and Smith normal form
computes their quotient in the ambient free abelian group.
\end{proof}

\begin{theorem}[Global localized structural rank of the
\(\rh/\cB_1\) equations]
\label{thm:global-rho}
For the complete minimum-circuit equation matrix \(E\),
\[
 I_{1312}(E)R_S=R_S.
\]
Consequently \(E\otimes_RR_S\) has rank exactly \(1312\) at every prime
of \(R_S\), and the residual \(\rh/\cB_1\) component equation
obstruction vanishes there.
\end{theorem}

\begin{proof}
The independently audited multiplication-lattice Smith form for
\([M(\delta_{\mathrm G})\ M(\delta_{\mathrm R})]\) has diagonal
\[
 \underbrace{17,\ldots,17}_{11\ \mathrm{times}},\qquad
 \underbrace{289,\ldots,289}_{5\ \mathrm{times}}.
\]
Therefore
\[
 [R:\delta_{\mathrm G}R+\delta_{\mathrm R}R]
 =17^{21}
 =69\,091\,933\,913\,008\,732\,880\,827\,217.
\]
Thus \((\delta_{\mathrm G},\delta_{\mathrm R})R[1/17]=R[1/17]\).
Both determinants belong to \(I_{1312}(E)\), although they use different
row and column selections.  Hence \(I_{1312}(E)R_S=R_S\) by
\cref{lem:determinantal-base-change}.  The structural kernel gives the
upper bound \(1312\), so the rank is exactly \(1312\).
\end{proof}

\begin{remark}[Which primes are inverted]
The determinantal ideal already becomes the unit ideal over \(R[1/17]\).
The additional inversion of \(2\) and \(3\) is used when identifying
component rank with the semisimple representation-theoretic obstruction,
because \(|\GammaG|=2^2\cdot3\cdot17\).
\end{remark}

\begin{remark}[Exact boundary]
\label{rem:global-rho-boundary}
\Cref{thm:global-rho} closes one of the six large component gates.  The
global gates for
\[
 \rh\om/\cB_1,\quad
 \rh/\cB_2,\quad
 \rh\om/\cB_2,\quad
 \rh/\cB_3,\quad
 \rh\om/\cB_3
\]
remain open.  The theorem does not prove
\(\operatorname{Supp}(\Theta_i)\subseteq\{2,3,17\}\), determine the
global Smith groups, or make a tomography or Markov-connectivity claim.
\end{remark}
