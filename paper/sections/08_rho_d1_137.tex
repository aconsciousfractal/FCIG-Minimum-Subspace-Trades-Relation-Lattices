\section{Complete saturation at three selected good primes}
\label{sec:selected-primes}

We first record the exact passage from complete component equations to the
integral quotient.

\begin{lemma}[Good-prime component detection]
\label{lem:good-prime}
Let \(p\nmid|\GammaG|\), suppose \(A_i\otimes\FF_p\) is surjective, and
let \(k/\FF_p\) split \(\GammaG\).  Put
\[
 Q_i=(K_i^{\ZZ}/L_i)\otimes k.
\]
For every absolutely irreducible \(k\GammaG\)-module \(V\), the
structural rank deficiency of the complete component equation map satisfies
\[
 \bigl(m_{B_i}(V)-m_P(V)\bigr)-\rank E_{i,V}
 =\dim_k\Hom_{k\GammaG}(Q_i,V).
\]
Consequently, if every complete component map has its structural target rank,
then
\(\Theta_i\otimes\FF_p=0\), and hence \(\Theta_i[p^\infty]=0\).
\end{lemma}

\begin{proof}
Reduction gives an exact sequence
\[
 0\longrightarrow Q_i
 \longrightarrow k[\cB_i]/(L_i\otimes k)
 \longrightarrow k[\cP]\longrightarrow0.
\]
Maschke's theorem makes \(k\GammaG\) semisimple.  Applying
\(\Hom_{k\GammaG}(-,V)\) gives
\[
 0\longrightarrow\Hom_{k\GammaG}(k[\cP],V)
 \longrightarrow\ker E_{i,V}
 \longrightarrow\Hom_{k\GammaG}(Q_i,V)
 \longrightarrow0.
\]
Taking dimensions gives the displayed deficiency formula.  If every absolute
type vanishes after a splitting extension, faithful scalar extension gives
\(Q_i=0\).
Finiteness of \(\Theta_i\) and Nakayama's lemma then give
\(\Theta_i[p^\infty]=0\).
\end{proof}

\subsection{The prime \(137\)}

Here \(137\nmid204\), every projector denominator is a unit, and the
incidence Smith form has no \(137\)-factor.  For \(\rh/\cB_1\), the block
multiplicity space has dimension \(1317\), while the point-induced
subspace has dimension \(5\), so the structural equation rank is
\[
 1317-5=1312.
\]
The complete circuit-equation matrix has a certified
\(1312\times1312\) minor with
\[
 \det=56\pmod {137}
 \qquad\text{at }\zeta_{17}=122.
\]
\begin{theorem}[Complete \(\rh/\cB_1\) contribution above \(137\)]
\label{thm:rho137}
The complete \(\rh/\cB_1\) equation matrix has structural rank \(1312\)
at every prime above \(137\).
\end{theorem}

\begin{proof}
The displayed minor is nonzero at \(\zeta_{17}=122\); the exact
certificates at the other fifteen roots likewise contain nonzero
order-\(1312\) minors.
\end{proof}

For \(\rh\om\), put
\[
 k_2=\FF_{137}[y]/(y^2+y+1).
\]
If \(r\) runs over the sixteen order-\(17\) roots in \(\FF_{137}\), then
\[
 \alpha_r=r^6y^2,\qquad
 \alpha_r^3=r,\qquad
 \alpha_r^{17}=y
\]
has order \(51\), and the Frobenius pairs
\(\{\alpha_r,\alpha_r^{137}\}\) represent the sixteen degree-two primes
of \(\ZZ[\zeta_{51}]\) above \(137\).  The six large-map certificates are
\[
\begin{array}{c|r|r|c|c}
\text{map}&\text{target rank}&\text{columns}&
 \text{residue degrees}&\text{nonzero minors}\\
\midrule
\rh/\cB_1&1312&1317&1^{16}&16/16\\
\rh\om/\cB_1&1305&10480&2^{16}&16/16\\
\rh/\cB_2&1314&5276&1^{16}&16/16\\
\rh\om/\cB_2&1304&10472&2^{16}&16/16\\
\rh/\cB_3&1310&5260&1^{16}&16/16\\
\rh\om/\cB_3&1306&10488&2^{16}&16/16.
\end{array}
\]
For quadratic residues, multiplication by \(a+by\) is expanded as
\[
 \begin{bmatrix}a&-b\\ b&a-b\end{bmatrix}.
\]
An independent parser uses Frobenius-conjugate roots, transposes every
selected square matrix, reproduces all \(80\) determinant residues for
the last five maps, and obtains zero after duplicating a field column.

\begin{theorem}[Complete \(137\)-primary quotients]
\label{thm:theta137}
For the three locked configurations,
\[
 \Theta_1[137^\infty]
 =\Theta_2[137^\infty]
 =\Theta_3[137^\infty]
 =0.
\]
\end{theorem}

\begin{proof}
\Cref{prop:small-components} excludes \(137\) from all eighteen small
component cokernels.  The table supplies structural rank for all six
large maps at every localization.  Apply \cref{lem:good-prime}.
\end{proof}

\subsection{The prime \(219\,097\)}

Since
\[
 219\,097-1=2^3\cdot3^2\cdot17\cdot179,
 \qquad 219\,097\equiv1\pmod {51},
\]
both cyclotomic orders split completely.  The complete
\(\rh/\cB_1\) equation matrix has a certified full-rank minor with
\[
 \det=213269\pmod {219097}
 \qquad\text{at }\zeta_{17}=91085.
\]
The complete large-map screen is
\[
\begin{array}{c|r|r|c}
\text{map}&\text{target rank}&\text{localizations}&\text{nonzero minors}\\
\midrule
\rh/\cB_1&1312&16&16\\
\rh\om/\cB_1&1305&32&32\\

\rh/\cB_3&1310&16&16\\

\end{array}
\]
For the last five maps, an independent parser enumerates the roots from a
primitive generator, reconstructs the transpose of every selected square
matrix, reproduces all \(128\) determinants, and obtains zero in all
\(128\) duplicate-column controls.

\begin{theorem}[Complete \(219\,097\)-primary quotients]
\label{thm:theta219097}
For the three locked configurations,
\[
 \Theta_1[219097^\infty]
 =\Theta_2[219097^\infty]

\begin{proof}

rank for the remaining six.  Apply \cref{lem:good-prime}.
\end{proof}

\subsection{The prime \(3\,288\,036\,131\)}

Put \(p=3\,288\,036\,131\).  Then
\[
 p-1=2\cdot5\cdot17\cdot2243\cdot8623,
 \qquad p\equiv1\pmod {17},\qquad p\equiv2\pmod3.
\]
Thus \(\Phi_{17}\) has sixteen linear factors and \(\Phi_{51}\) has
sixteen irreducible quadratic factors.  The complete
\(\rh/\cB_1\) equation matrix has a certified full-rank minor with
\[
 \det=1\,471\,654\,912\pmod p
 \qquad\text{at }\zeta_{17}=1\,120\,906\,769.
\]
For the \(\rh\om\)-maps, use
\[
 k_2=\FF_p[y]/(y^2+y+1),\qquad \alpha_r=r^6y^2,
\]
where \(r\) runs through the order-\(17\) roots.  The complete screen is
\[
\begin{array}{c|r|r|c|c}
\text{map}&\text{target rank}&\text{residue degree}&
 \text{localizations}&\text{nonzero minors}\\
\midrule
\rh/\cB_1&1312&1&16&16\\
\rh\om/\cB_1&1305&2&16&16\\
\rh/\cB_2&1314&1&16&16\\
\rh\om/\cB_2&1304&2&16&16\\
\rh/\cB_3&1310&1&16&16\\
\rh\om/\cB_3&1306&2&16&16.
\end{array}
\]
The independent route finds the least primitive generator \(2\), uses
Frobenius-conjugate quadratic representatives, transposes every selected
square matrix, reproduces all \(80\) determinant residues for the last
five maps, and obtains zero in all \(80\) duplicate-column controls.

\begin{theorem}[Complete \(3\,288\,036\,131\)-primary quotients]
\label{thm:theta3288036131}
For the three locked configurations,
\[
 \Theta_1[p^\infty]=\Theta_2[p^\infty]=\Theta_3[p^\infty]=0.
\]
\end{theorem}

\begin{proof}
\Cref{prop:small-components} excludes \(p\) from all eighteen small
components, and the complete large-map screen supplies structural rank
for the remaining six.  Apply \cref{lem:good-prime}.
\end{proof}

\begin{remark}[Exact boundary]
The three theorems classify the complete primary quotients only at the
displayed rational primes.  A finite list of local screens does not
determine the support or Smith factors of the global groups
\(\Theta_i\).  The next section gives a global determinantal-ideal
certificate for the single \(\rh/\cB_1\) equation module.
\end{remark}
